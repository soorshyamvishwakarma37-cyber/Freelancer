🔥 Firebase Setup — 5 Steps
  1. Project: console.firebase.google.com → Add Project `sewaastra`
  2. Web App: `</>` → Copy `firebaseConfig` → इस File में Paste (पहले से लगा है)
  3. Auth: Authentication → Google → Enable
  4. Firestore + Storage: Create (Test mode, asia-south1)
rules_version='2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /bookings/{id}{allow create:if true; allow read:if true; allow update,delete:if true;}
    match /rooms/{id}{allow read:if true; allow create,update,delete:if true;}
    match /gallery/{id}{allow read:if true; allow create,update,delete:if true;}
    match /physicalRooms/{id}{allow read:if true; allow write:if true;}
    match /settings/{id}{allow read:if true; allow write:if request.auth!=null && request.auth.token.email=="[email protected]";}
    match /staffAccounts/{id}{allow read:if true; allow write:if request.auth!=null && request.auth.token.email=="[email protected]";}
    match /staffSessions/{id}{allow read:if true; allow write:if true;}
    match /reviews/{id}{allow read:if true; allow create:if true; allow update:if true; allow delete:if request.auth!=null && request.auth.token.email=="[email protected]";}
    match /serviceRequests/{id}{allow create:if true; allow read,delete:if request.auth!=null && request.auth.token.email=="[email protected]";}
    match /menuItems/{id}{allow read:if true; allow create,update,delete:if true;}
    match /staffOtps/{id}{allow read,create,update,delete:if true;}
  }
}
